cmake_minimum_required(VERSION 3.28)
project(OneByteAudioObs VERSION 0.4.1 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 only")
endif()
find_package(libobs CONFIG REQUIRED)
set(AUDIO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../1byte-audio-vst2")
set(SEND_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../zosma-send")
set(NDI_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/ndi/include")
add_library(onebyte-audio-obs MODULE source.cpp "${AUDIO_ROOT}/ndi_audio.cpp")
target_include_directories(onebyte-audio-obs PRIVATE "${AUDIO_ROOT}" "${SEND_ROOT}" "${NDI_HEADERS}")
target_compile_features(onebyte-audio-obs PRIVATE cxx_std_17)
target_compile_definitions(onebyte-audio-obs PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
    _WIN32_WINNT=0x0A00 PROCESSINGNDILIB_STATIC NDILIB_CPP_DEFAULT_CONSTRUCTORS=0 ONEBYTE_NDI_SUBDIR)
target_link_libraries(onebyte-audio-obs PRIVATE OBS::libobs avrt)
set_target_properties(onebyte-audio-obs PROPERTIES PREFIX "" OUTPUT_NAME "1byte-labs-audio")
add_executable(obs-source-test tests/source_test.cpp "${AUDIO_ROOT}/ndi_audio.cpp")
target_include_directories(obs-source-test PRIVATE "${AUDIO_ROOT}" "${SEND_ROOT}" "${NDI_HEADERS}")
target_compile_features(obs-source-test PRIVATE cxx_std_17)
target_compile_definitions(obs-source-test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
    _WIN32_WINNT=0x0A00 PROCESSINGNDILIB_STATIC NDILIB_CPP_DEFAULT_CONSTRUCTORS=0 ONEBYTE_NDI_SUBDIR)
target_link_libraries(obs-source-test PRIVATE OBS::libobs avrt)
if(MSVC)
    foreach(target onebyte-audio-obs obs-source-test)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
    endforeach()
endif()
