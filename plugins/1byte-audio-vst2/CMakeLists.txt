cmake_minimum_required(VERSION 3.24)
project(OneByteAudioVst2 VERSION 0.4.1 LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
set(SEND_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../zosma-send")
add_executable(vst2-queue-test "${SEND_ROOT}/tests/audio_queue_test.cpp")
target_compile_features(vst2-queue-test PRIVATE cxx_std_17)
target_link_libraries(vst2-queue-test PRIVATE Threads::Threads)
add_test(NAME queue COMMAND vst2-queue-test)
add_executable(vst2-gain-test "${SEND_ROOT}/tests/send_gain_test.cpp")
target_compile_features(vst2-gain-test PRIVATE cxx_std_17)
add_test(NAME gain COMMAND vst2-gain-test)
option(ONEBYTE_PORTABLE_TESTS_ONLY "Build queue/gain tests without Windows" OFF)
if(ONEBYTE_PORTABLE_TESTS_ONLY)
    return()
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "1Byte Labs Audio requires Windows x64")
endif()
include(FetchContent)
# Independently implemented ABI, BSD-3-Clause; no proprietary VST2 SDK.
FetchContent_Declare(vst2abi
    GIT_REPOSITORY https://github.com/Xaymar/vst2sdk.git
    GIT_TAG 339d4f31590bf77c0d0d248e09a380ac6285e069)
FetchContent_GetProperties(vst2abi)
if(NOT vst2abi_POPULATED)
    FetchContent_Populate(vst2abi)
endif()
set(NDI_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/ndi/include")
set(COMMON_DEFINITIONS UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00
    PROCESSINGNDILIB_STATIC NDILIB_CPP_DEFAULT_CONSTRUCTORS=0)
add_library(onebyte-audio MODULE plugin.cpp editor.cpp ndi_audio.cpp)
target_include_directories(onebyte-audio PRIVATE "${vst2abi_SOURCE_DIR}/include" "${SEND_ROOT}" "${NDI_HEADERS}")
target_compile_features(onebyte-audio PRIVATE cxx_std_17)
target_compile_definitions(onebyte-audio PRIVATE ${COMMON_DEFINITIONS})
target_link_libraries(onebyte-audio PRIVATE avrt user32 gdi32 uxtheme comctl32 shell32)
set_target_properties(onebyte-audio PROPERTIES PREFIX "" OUTPUT_NAME "1Byte Labs Audio")

add_library(mock-ndi MODULE tests/mock_ndi.cpp)
target_include_directories(mock-ndi PRIVATE "${NDI_HEADERS}")
target_compile_definitions(mock-ndi PRIVATE ${COMMON_DEFINITIONS})
target_compile_features(mock-ndi PRIVATE cxx_std_17)
set_target_properties(mock-ndi PROPERTIES PREFIX "" OUTPUT_NAME "Processing.NDI.Lib.x64")
if(MSVC)
    target_link_options(mock-ndi PRIVATE /EXPORT:NDIlib_v6_load)
endif()
add_executable(vst2-host-test tests/host_test.cpp)
target_include_directories(vst2-host-test PRIVATE "${vst2abi_SOURCE_DIR}/include")
target_compile_definitions(vst2-host-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_features(vst2-host-test PRIVATE cxx_std_17)
target_link_libraries(vst2-host-test PRIVATE user32 gdi32 comctl32)
add_dependencies(vst2-host-test onebyte-audio mock-ndi)
add_test(NAME vst2-host COMMAND vst2-host-test "$<TARGET_FILE:onebyte-audio>" "$<TARGET_FILE:mock-ndi>")
if(MSVC)
    foreach(target onebyte-audio mock-ndi vst2-host-test)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
    endforeach()
endif()

add_executable(vst2-diagnostic-test tests/diagnostic_test.cpp ndi_audio.cpp)
target_include_directories(vst2-diagnostic-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${SEND_ROOT}" "${NDI_HEADERS}")
target_compile_features(vst2-diagnostic-test PRIVATE cxx_std_17)
target_compile_definitions(vst2-diagnostic-test PRIVATE ${COMMON_DEFINITIONS})
target_link_libraries(vst2-diagnostic-test PRIVATE avrt)
add_dependencies(vst2-diagnostic-test mock-ndi)
if(MSVC)
    target_compile_options(vst2-diagnostic-test PRIVATE /W4 /utf-8 /EHsc)
endif()
add_test(NAME ndi-diagnostics COMMAND vst2-diagnostic-test)
# A separate process/directory ensures no loaded runtime can hide a missing DLL.
add_custom_command(TARGET vst2-diagnostic-test POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:vst2-diagnostic-test>/missing-runtime"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:vst2-diagnostic-test>" "$<TARGET_FILE_DIR:vst2-diagnostic-test>/missing-runtime/")
add_test(NAME ndi-missing-runtime COMMAND "$<TARGET_FILE_DIR:vst2-diagnostic-test>/missing-runtime/vst2-diagnostic-test.exe" missing)

add_executable(vst2-receive-loopback tests/receive_loopback_test.cpp ndi_audio.cpp)
target_include_directories(vst2-receive-loopback PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${SEND_ROOT}" "${NDI_HEADERS}")
target_compile_features(vst2-receive-loopback PRIVATE cxx_std_17)
target_compile_definitions(vst2-receive-loopback PRIVATE ${COMMON_DEFINITIONS})
target_link_libraries(vst2-receive-loopback PRIVATE avrt)
if(MSVC)
    target_compile_options(vst2-receive-loopback PRIVATE /W4 /utf-8 /EHsc)
endif()
